```go
package value

import (
	"math"
	"time"

	"github.com/tunarider/fung/internal/diag"
)

// Time is an instant with its time zone (02). It has no literal form:
// time.now(), time.parse and time.from_unix make it.
type Time struct{ T time.Time }

func NewTime(t time.Time) Time { return Time{T: t} }

func (Time) TypeName() string { return "time" }

// ISO renders the value as RFC 3339 with fractional seconds only when
// they are not zero (06: 시간 형식).
func (t Time) ISO() string { return t.T.Format(time.RFC3339Nano) }

// addTimeDur is time + dur / time - dur (07).
func addTimeDur(t Time, d Dur) (Value, *diag.Error) {
	r := t.T.Add(time.Duration(d))
	if (d > 0 && !r.After(t.T)) || (d < 0 && !r.Before(t.T)) {
		return nil, overflow()
	}
	return Time{T: r}, nil
}

// subTimes is time - time → dur. Go's Sub saturates at the duration
// range; that is an overflow here (07: 범위를 넘으면 에러).
func subTimes(a, b Time) (Value, *diag.Error) {
	d := a.T.Sub(b.T)
	if d == time.Duration(math.MaxInt64) || d == time.Duration(math.MinInt64) {
		return nil, overflow()
	}
	return Dur(d), nil
}
```

```go
func TestRunTaskRunsOnAFreshTask(t *testing.T) {

		Params: []value.NativeParam{{Name: "f"}},

			}
```

```go
// Context is what a native function may ask the VM for.
type Context interface {
	Stdout() io.Writer
	Stderr() io.Writer
	ReadLine() (string, error)                  // one line of standard input (io.EOF at the end), shared by all tasks
	Call(fn Value, args []Value) (Value, error) // re-enter the VM (map, at_exit, …)
	AtExit(fn Value)                            // register an exit handler
	Done() <-chan struct{}                      // this task's cancellation signal (nil: never)
	RunTask(fn Value, args []Value) (Value, error) // run fn on a fresh child task, on this goroutine, and wait for it
}
```

```go
// Done implements value.Context: the task's cancellation channel, so a
// native blocked in Go (an HTTP server or request) can stop with it.
func (t *Task) Done() <-chan struct{} { return t.done() }

// RunTask implements value.Context: it runs fn on a fresh child task on
// the calling goroutine (an HTTP handler, 06) and returns its result.
// The child counts as live for the deadlock detector and inherits this
// task's cancellation.
func (t *Task) RunTask(fn value.Value, args []value.Value) (value.Value, error) {
	child := t.rt.newTask(t.ctxOrRoot())
	defer child.cancel()
	var v value.Value
	var err error
	t.rt.runLive(func() { v, err = child.Call(fn, args) })
	return v, err
}
```

```go
package native

import (
	"fmt"
	"strconv"
	"strings"
	"time"
)

// tokens are the time format tokens (06: 시간 형식), longest first so
// YYYY wins over YY and ZZ over Z. Characters outside the table are
// literal: they print as they are and must match on parse.
var tokens = []string{"YYYY", "SSS", "YY", "MM", "DD", "HH", "mm", "ss", "ZZ", "Z"}

// nextToken returns the token at the start of layout, or "" for a
// literal character.
func nextToken(layout string) string {
	for _, tk := range tokens {
		if strings.HasPrefix(layout, tk) {
			return tk
		}
	}
	return ""
}

func pad(n, width int) string {
	s := strconv.Itoa(n)
	for len(s) < width {
		s = "0" + s
	}
	return s
}

func zone(t time.Time, colon bool) string {
	_, off := t.Zone()
	sign := "+"
	if off < 0 {
		sign, off = "-", -off
	}
	h, m := off/3600, (off%3600)/60
	if colon {
		return sign + pad(h, 2) + ":" + pad(m, 2)
	}
	return sign + pad(h, 2) + pad(m, 2)
}

// formatTime renders t with the token layout.
func formatTime(t time.Time, layout string) string {
	var b strings.Builder
	for layout != "" {
		tk := nextToken(layout)
		switch tk {
		case "YYYY":
			b.WriteString(pad(t.Year(), 4))
		case "YY":
			b.WriteString(pad(t.Year()%100, 2))
		case "MM":
			b.WriteString(pad(int(t.Month()), 2))
		case "DD":
			b.WriteString(pad(t.Day(), 2))
		case "HH":
			b.WriteString(pad(t.Hour(), 2))
		case "mm":
			b.WriteString(pad(t.Minute(), 2))
		case "ss":
			b.WriteString(pad(t.Second(), 2))
		case "SSS":
			b.WriteString(pad(t.Nanosecond()/1_000_000, 3))
		case "Z":
			b.WriteString(zone(t, true))
		case "ZZ":
			b.WriteString(zone(t, false))
		default:
			r := []rune(layout)[0]
			b.WriteRune(r)
			layout = layout[len(string(r)):]
			continue
		}
		layout = layout[len(tk):]
	}
	return b.String()
}

// digits reads exactly n ASCII digits from the front of s.
func digits(s string, n int, what string) (int, string, error) {
	if len(s) < n {
		return 0, "", fmt.Errorf("expected %d digits for %s", n, what)
	}
	for i := 0; i < n; i++ {
		if s[i] < '0' || s[i] > '9' {
			return 0, "", fmt.Errorf("expected %d digits for %s", n, what)
		}
	}
	v, _ := strconv.Atoi(s[:n])
	return v, s[n:], nil
}

// parseZone reads +HH:MM (colon) or +HHMM and returns the offset in seconds.
func parseZone(s string, colon bool) (int, string, error) {
	if s == "" || (s[0] != '+' && s[0] != '-') {
		return 0, "", fmt.Errorf("expected a time zone offset")
	}
	sign := 1
	if s[0] == '-' {
		sign = -1
	}
	h, rest, err := digits(s[1:], 2, "the zone hours")
	if err != nil {
		return 0, "", err
	}
	if colon {
		if !strings.HasPrefix(rest, ":") {
			return 0, "", fmt.Errorf("expected `:` in the time zone offset")
		}
		rest = rest[1:]
	}
	m, rest, err := digits(rest, 2, "the zone minutes")
	if err != nil {
		return 0, "", err
	}
	return sign * (h*3600 + m*60), rest, nil
}

// parseTime reads s with the token layout. Fields the layout does not
// mention are zero (year 0 → 0000 is accepted as Go does); without a
// zone token the time is local. Out-of-range fields (month 13, Feb 30)
// are errors: time.Date would normalize them silently.
func parseTime(s, layout string) (time.Time, error) {
	year, month, day, hour, min, sec, ms := 0, 1, 1, 0, 0, 0, 0
	var loc *time.Location = time.Local
	var err error
	for layout != "" {
		tk := nextToken(layout)
		switch tk {
		case "YYYY":
			year, s, err = digits(s, 4, "the year")
		case "YY":
			year, s, err = digits(s, 2, "the year")
			year += 2000
		case "MM":
			month, s, err = digits(s, 2, "the month")
		case "DD":
			day, s, err = digits(s, 2, "the day")
		case "HH":
			hour, s, err = digits(s, 2, "the hour")
		case "mm":
			min, s, err = digits(s, 2, "the minute")
		case "ss":
			sec, s, err = digits(s, 2, "the second")
		case "SSS":
			ms, s, err = digits(s, 3, "the milliseconds")
		case "Z", "ZZ":
			var off int
			off, s, err = parseZone(s, tk == "Z")
			loc = time.FixedZone("", off)
		default:
			r := []rune(layout)[0]
			if !strings.HasPrefix(s, string(r)) {
				return time.Time{}, fmt.Errorf("expected %q", string(r))
			}
			s = s[len(string(r)):]
			layout = layout[len(string(r)):]
			continue
		}
		if err != nil {
			return time.Time{}, err
		}
		layout = layout[len(tk):]
	}
	if s != "" {
		return time.Time{}, fmt.Errorf("unexpected text %q after the date", s)
	}
	if month < 1 || month > 12 || hour > 23 || min > 59 || sec > 59 {
		return time.Time{}, fmt.Errorf("field out of range")
	}
	t := time.Date(year, time.Month(month), day, hour, min, sec, ms*1_000_000, loc)
	if t.Day() != day || int(t.Month()) != month {
		return time.Time{}, fmt.Errorf("day %d is out of range for the month", day)
	}
	return t, nil
}
```

```go
	"time"
```

```go
// timeModule builds _native.time (06): instants and the token formats.
func timeModule() *value.Module {
	m := value.NewModule("time", "_native.time")
	add := func(n *value.Native) { m.Set(n.Name, n) }
	add(nat.Def("now", timeNowFn))
	add(nat.Def("parse", timeParseFn, nat.Param("s"), nat.Opt("format", value.Nil{})))
	add(nat.Def("format", timeFormatFn, nat.Param("t"), nat.Opt("format", value.Nil{})))
	add(nat.Def("utc", timeUTCFn, nat.Param("t")))
	add(nat.Def("local", timeLocalFn, nat.Param("t")))
	add(nat.Def("unix", timeUnixFn, nat.Param("t")))
	add(nat.Def("from_unix", timeFromUnixFn, nat.Param("n")))
	add(nat.Def("parts", timePartsFn, nat.Param("t")))
	return m
}

func needTime(name string, v value.Value) (time.Time, error) {
	t, ok := v.(value.Time)
	if !ok {
		return time.Time{}, diag.Errorf("E0403", diag.Pos{}, name, "time", v.TypeName())
	}
	return t.T, nil
}

func timeNowFn(_ value.Context, _ []value.Value) (value.Value, error) {
	return value.NewTime(time.Now()), nil
}

func timeParseFn(_ value.Context, args []value.Value) (value.Value, error) {
	s, err := nat.NeedStr("s", args[0])
	if err != nil {
		return nil, err
	}
	layout, has, err := nat.OptStr("format", args[1])
	if err != nil {
		return nil, err
	}
	var t time.Time
	var perr error
	if has {
		t, perr = parseTime(s, layout)
	} else {
		t, perr = time.Parse(time.RFC3339Nano, s)
	}
	if perr != nil {
		return nat.ErrValue("parse", "cannot parse `"+s+"` as a time: "+perr.Error()), nil
	}
	return nat.Ok(value.NewTime(t)), nil
}

func timeFormatFn(_ value.Context, args []value.Value) (value.Value, error) {
	t, err := needTime("t", args[0])
	if err != nil {
		return nil, err
	}
	layout, has, err := nat.OptStr("format", args[1])
	if err != nil {
		return nil, err
	}
	if !has {
		return value.Str(value.NewTime(t).ISO()), nil
	}
	return value.Str(formatTime(t, layout)), nil
}

func timeUTCFn(_ value.Context, args []value.Value) (value.Value, error) {
	t, err := needTime("t", args[0])
	if err != nil {
		return nil, err
	}
	return value.NewTime(t.UTC()), nil
}

func timeLocalFn(_ value.Context, args []value.Value) (value.Value, error) {
	t, err := needTime("t", args[0])
	if err != nil {
		return nil, err
	}
	return value.NewTime(t.Local()), nil
}

func timeUnixFn(_ value.Context, args []value.Value) (value.Value, error) {
	t, err := needTime("t", args[0])
	if err != nil {
		return nil, err
	}
	return value.Int(t.Unix()), nil
}

func timeFromUnixFn(_ value.Context, args []value.Value) (value.Value, error) {
	n, err := nat.NeedInt("n", args[0])
	if err != nil {
		return nil, err
	}
	return value.NewTime(time.Unix(int64(n), 0)), nil
}

var weekdays = [...]string{"sun", "mon", "tue", "wed", "thu", "fri", "sat"}

func timePartsFn(_ value.Context, args []value.Value) (value.Value, error) {
	t, err := needTime("t", args[0])
	if err != nil {
		return nil, err
	}
	m := value.NewMap()
	m.Set("year", value.Int(t.Year()))
	m.Set("month", value.Int(int(t.Month())))
	m.Set("day", value.Int(t.Day()))
	m.Set("hour", value.Int(t.Hour()))
	m.Set("minute", value.Int(t.Minute()))
	m.Set("second", value.Int(t.Second()))
	m.Set("weekday", value.Intern(weekdays[t.Weekday()]))
	return m, nil
}
```

```go
// jsonModule builds _native.json (06): a token-level decoder that keeps
// object key order, and an encoder over fung values. It lives in Go
// because parsing walks every character (06: 성능이 이유면 native).
func jsonModule() *value.Module {
	m := value.NewModule("json", "_native.json")
	add := func(n *value.Native) { m.Set(n.Name, n) }
	add(nat.Def("parse", jsonParseFn, nat.Param("s")))
	add(nat.Def("stringify", jsonStringifyFn, nat.Param("v"), nat.Opt("indent", value.Nil{})))
	return m

// jsonError is a parse failure with a 1-based position.
type jsonError struct {
	line, col int
	msg       string
}

func (e *jsonError) Error() string { return e.msg }

// lineCol converts a byte offset into a 1-based line and column.
func lineCol(s string, off int64) (int, int) {
	if off > int64(len(s)) {
		off = int64(len(s))
	}
	before := s[:off]
	line := strings.Count(before, "\n") + 1
	col := len([]rune(before[strings.LastIndex(before, "\n")+1:])) + 1
	return line, col
}

func jsonParseFn(_ value.Context, args []value.Value) (value.Value, error) {
	s, err := nat.NeedStr("s", args[0])
	if err != nil {
		return nil, err
	}
	v, perr := decodeJSON(s)
	if perr != nil {
		je := perr.(*jsonError)
		return nat.ErrValue("parse", fmt.Sprintf("invalid JSON at line %d, column %d: %s", je.line, je.col, je.msg),
			value.Str("line"), value.Int(je.line), value.Str("col"), value.Int(je.col)), nil
	}
	return nat.Ok(v), nil
}

// decodeJSON reads one JSON value and insists nothing follows it.
func decodeJSON(s string) (value.Value, error) {
	dec := json.NewDecoder(strings.NewReader(s))
	dec.UseNumber()
	v, err := readValue(dec, s)
	if err != nil {
		return nil, err
	}
	if _, err := dec.Token(); err != io.EOF {
		return nil, jsonErr(s, dec.InputOffset(), err, "unexpected text after the value")
	}
	return v, nil
}

// jsonErr positions a decoder error: a syntax error carries its own
// offset, anything else uses the decoder's current one.
func jsonErr(s string, off int64, err error, fallback string) error {
	msg := fallback
	var se *json.SyntaxError
	if errors.As(err, &se) {
		off, msg = se.Offset, se.Error()
	} else if err != nil && err != io.EOF {
		msg = err.Error()
	}
	line, col := lineCol(s, off)
	return &jsonError{line: line, col: col, msg: msg}
}

func readValue(dec *json.Decoder, s string) (value.Value, error) {
	start := dec.InputOffset()
	tok, err := dec.Token()
	if err != nil {
		if err == io.EOF {
			return nil, jsonErr(s, start, nil, "unexpected end of input")
		}
		return nil, jsonErr(s, start, err, "invalid token")
	}
	switch x := tok.(type) {
	case json.Delim:
		switch x {
		case '{':
			m := value.NewMap()
			for dec.More() {
				kt, err := dec.Token()
				if err != nil {
					return nil, jsonErr(s, dec.InputOffset(), err, "invalid object key")
				}
				k, ok := kt.(string)
				if !ok {
					return nil, jsonErr(s, dec.InputOffset(), nil, "object keys must be strings")
				}
				v, err := readValue(dec, s)
				if err != nil {
					return nil, err
				}
				m.Set(k, v)
			}
			if _, err := dec.Token(); err != nil { // the closing }
				return nil, jsonErr(s, dec.InputOffset(), err, "unterminated object")
			}
			return m, nil
		case '[':
			items := []value.Value{}
			for dec.More() {
				v, err := readValue(dec, s)
				if err != nil {
					return nil, err
				}
				items = append(items, v)
			}
			if _, err := dec.Token(); err != nil { // the closing ]
				return nil, jsonErr(s, dec.InputOffset(), err, "unterminated array")
			}
			return &value.List{Items: items}, nil
		}
		return nil, jsonErr(s, start, nil, "unexpected "+string(x))
	case string:
		return value.Str(x), nil
	case bool:
		return value.Bool(x), nil
	case nil:
		return value.Nil{}, nil
	case json.Number:
		return numberValue(string(x), s, start)
	}
	return nil, jsonErr(s, start, nil, "unexpected token")
}

// numberValue keeps integers exact: a literal without `.`, `e` or `E`
// is an int and must fit int64 (06: 조용히 float로 바꾸지 않는다).
func numberValue(lit, s string, off int64) (value.Value, error) {
	if !strings.ContainsAny(lit, ".eE") {
		n, err := strconv.ParseInt(lit, 10, 64)
		if err != nil {
			return nil, jsonErr(s, off, nil, "integer "+lit+" is out of range")
		}
		return value.Int(n), nil
	}
	f, err := strconv.ParseFloat(lit, 64)
	if err != nil {
		return nil, jsonErr(s, off, nil, "number "+lit+" is out of range")
	}
	return value.Float(f), nil
}

func jsonStringifyFn(_ value.Context, args []value.Value) (value.Value, error) {
	indent := 0
	switch x := args[1].(type) {
	case value.Nil:
	case value.Int:
		if x < 0 {
			return nil, diag.Errorf("E0403", diag.Pos{}, "indent", "a non-negative int", value.Format(x))
		}
		indent = int(x)
	default:
		return nil, diag.Errorf("E0403", diag.Pos{}, "indent", "int | nil", args[1].TypeName())
	}
	var b bytes.Buffer
	if err := encodeJSON(&b, args[0], indent, 0); err != nil {
		return nil, err
	}
	return value.Str(b.String()), nil
}

// quoteJSON escapes a string the JSON way without HTML escaping.
func quoteJSON(s string) string {
	var b bytes.Buffer
	enc := json.NewEncoder(&b)
	enc.SetEscapeHTML(false)
	enc.Encode(s) // cannot fail for a string
	return strings.TrimSuffix(b.String(), "\n")
}

// encodeJSON writes v (06: stringify 대응표). Maps keep their order.
// indent 0 is one line; otherwise nested values go on their own lines.
func encodeJSON(b *bytes.Buffer, v value.Value, indent, depth int) error {
	newline := func(d int) {
		if indent > 0 {
			b.WriteByte('\n')
			b.WriteString(strings.Repeat(" ", indent*d))
		}
	}
	colon := ":"
	if indent > 0 {
		colon = ": "
	}
	switch x := v.(type) {
	case value.Nil:
		b.WriteString("null")
	case value.Bool, value.Int, value.Float:
		b.WriteString(value.Format(x))
	case value.Str:
		b.WriteString(quoteJSON(string(x)))
	case value.Sym:
		b.WriteString(quoteJSON(x.Name()))
	case value.Time:
		b.WriteString(quoteJSON(x.ISO()))
	case *value.List:
		return encodeItems(b, x.Items, indent, depth, newline)
	case *value.Tuple:
		return encodeItems(b, x.Items, indent, depth, newline)
	case *value.Map:
		if x.Len() == 0 {
			b.WriteString("{}")
			return nil
		}
		b.WriteByte('{')
		for i, k := range x.Keys() {
			if i > 0 {
				b.WriteByte(',')
			}
			newline(depth + 1)
			b.WriteString(quoteJSON(k))
			b.WriteString(colon)
			val, _ := x.Get(k)
			if err := encodeJSON(b, val, indent, depth+1); err != nil {
				return err
			}
		}
		newline(depth)
		b.WriteByte('}')
	default:
		return diag.Errorf("E0513", diag.Pos{}, v.TypeName(), "JSON")
	}
	return nil
}

func encodeItems(b *bytes.Buffer, items []value.Value, indent, depth int, newline func(int)) error {
	if len(items) == 0 {
		b.WriteString("[]")
		return nil
	}
	b.WriteByte('[')
	for i, it := range items {
		if i > 0 {
			b.WriteByte(',')
		}
		newline(depth + 1)
		if err := encodeJSON(b, it, indent, depth+1); err != nil {
			return err
		}
	}
	newline(depth)
	b.WriteByte(']')
	return nil
}
```

```go
// regexModule builds _native.regex (06): Go RE2 over fung strings. Match
// positions are rune indexes (02: 문자열은 글자 단위), not byte offsets.
func regexModule() *value.Module {
	m := value.NewModule("regex", "_native.regex")
	add := func(n *value.Native) { m.Set(n.Name, n) }
	add(nat.Def("match", reMatchFn, nat.Param("s"), nat.Param("pat")))
	add(nat.Def("find", reFindFn, nat.Param("s"), nat.Param("pat")))
	add(nat.Def("find_all", reFindAllFn, nat.Param("s"), nat.Param("pat")))
	add(nat.Def("replace", reReplaceFn, nat.Param("s"), nat.Param("pat"), nat.Param("repl")))
	add(nat.Def("split", reSplitFn, nat.Param("s"), nat.Param("pat")))
	return m
}

// maxCachedPatterns bounds the compiled-pattern cache; past it the cache
// is emptied rather than evicted one by one (patterns are few and small).
const maxCachedPatterns = 256

var (
	patternMu    sync.Mutex
	patternCache = map[string]*regexp.Regexp{}
)

// compilePattern returns the compiled pattern, cached. A bad pattern is
// a code error (E0515): patterns are written in the script.
func compilePattern(pat string) (*regexp.Regexp, error) {
	patternMu.Lock()
	defer patternMu.Unlock()
	if re, ok := patternCache[pat]; ok {
		return re, nil
	}
	re, err := regexp.Compile(pat)
	if err != nil {
		return nil, diag.Errorf("E0515", diag.Pos{}, err.Error())
	}
	if len(patternCache) >= maxCachedPatterns {
		patternCache = map[string]*regexp.Regexp{}
	}
	patternCache[pat] = re
	return re, nil
}

// subjectAndPattern reads the (s, pat) arguments every function takes.
func subjectAndPattern(args []value.Value) (string, *regexp.Regexp, error) {
	s, err := nat.NeedStr("s", args[0])
	if err != nil {
		return "", nil, err
	}
	pat, err := nat.NeedStr("pat", args[1])
	if err != nil {
		return "", nil, err
	}
	re, err := compilePattern(pat)
	return s, re, err
}

// runeIndex converts a byte offset in s to a rune index.
func runeIndex(s string, byteOff int) int { return utf8.RuneCountInString(s[:byteOff]) }

// matchValue builds {text, start, end, groups, named} from submatch
// byte offsets (pairs, -1 for a group that did not take part).
func matchValue(s string, re *regexp.Regexp, loc []int) value.Value {
	m := value.NewMap()
	m.Set("text", value.Str(s[loc[0]:loc[1]]))
	m.Set("start", value.Int(runeIndex(s, loc[0])))
	m.Set("end", value.Int(runeIndex(s, loc[1])))
	groups := []value.Value{}
	named := value.NewMap()
	names := re.SubexpNames()
	for i := 1; i < len(loc)/2; i++ {
```

```go
	add(nat.Def("request", httpRequestFn, nat.Param("method"), nat.Param("url"), nat.Opt("body", value.Nil{}), nat.Opt("headers", value.NewMap()), nat.Opt("timeout", defTimeout)))
	return m

// taskContext derives a Go context from the task's cancellation signal
// and an optional timeout (0: none).
func taskContext(done <-chan struct{}, timeout time.Duration) (context.Context, context.CancelFunc) {
	ctx, cancel := context.WithCancel(context.Background())
	if timeout > 0 {
		ctx, cancel = context.WithTimeout(context.Background(), timeout)
	}
	if done != nil {
		// Ends when the task is cancelled or when cancel() is called, so
		// the goroutine never outlives the request.
		go func() {
			select {
			case <-done:
				cancel()
			case <-ctx.Done():
			}
		}()
	}
	return ctx, cancel
}

// headerMap lowers header names and joins repeated values (06).
func headerMap(h http.Header) *value.Map {
	keys := make([]string, 0, len(h))
	for k := range h {
		keys = append(keys, k)
	}
	sort.Strings(keys)
	m := value.NewMap()
	for _, k := range keys {
		m.Set(strings.ToLower(k), value.Str(strings.Join(h[k], ", ")))
	}
	return m
}

// requestArgs reads the shared (body, headers, timeout) arguments.
func requestArgs(body, headers, timeout value.Value) (string, bool, *value.Map, time.Duration, error) {
	b, hasBody, err := nat.OptStr("body", body)
	if err != nil {
		return "", false, nil, 0, err
	}
	h, ok := headers.(*value.Map)
	if !ok {
		return "", false, nil, 0, diag.Errorf("E0403", diag.Pos{}, "headers", "map", headers.TypeName())
	}
	d, ok := timeout.(value.Dur)
	if !ok {
		return "", false, nil, 0, diag.Errorf("E0403", diag.Pos{}, "timeout", "dur", timeout.TypeName())
	}
	return b, hasBody, h, time.Duration(d), nil
}

func httpGetFn(ctx value.Context, args []value.Value) (value.Value, error) {
	return doRequest(ctx, "GET", args[0], value.Nil{}, args[1], args[2])
}

func httpPostFn(ctx value.Context, args []value.Value) (value.Value, error) {
	return doRequest(ctx, "POST", args[0], args[1], args[2], args[3])
}

func httpRequestFn(ctx value.Context, args []value.Value) (value.Value, error) {
	method, err := nat.NeedStr("method", args[0])
	if err != nil {
		return nil, err
	}
	return doRequest(ctx, strings.ToUpper(method), args[1], args[2], args[3], args[4])
}

// doRequest performs one request. Any response is Ok (06: 404도 Ok);
// transport failures are Err 'network, deadlines Err 'timeout, and a
// cancelled task unwinds with E0701 like every blocking primitive.
func doRequest(ctx value.Context, method string, urlV, bodyV, headersV, timeoutV value.Value) (value.Value, error) {
	url, err := nat.NeedStr("url", urlV)
	if err != nil {
		return nil, err
	}
	body, hasBody, headers, timeout, err := requestArgs(bodyV, headersV, timeoutV)
	if err != nil {
		return nil, err
	}
	var reader io.Reader
	if hasBody {
		reader = strings.NewReader(body)
	}
	gctx, cancel := taskContext(ctx.Done(), timeout)
	defer cancel()
	req, rerr := http.NewRequestWithContext(gctx, method, url, reader)
	if rerr != nil {
		return nat.ErrValue("network", "cannot request `"+url+"`: "+rerr.Error()), nil
	}
	for _, k := range headers.Keys() {
		v, _ := headers.Get(k)
		s, ok := v.(value.Str)
		if !ok {
			return nil, diag.Errorf("E0403", diag.Pos{}, "headers", "a map of strings", "a map holding "+v.TypeName())
		}
		req.Header.Set(k, string(s))
	}
	resp, derr := http.DefaultClient.Do(req)
	if derr != nil {
		return requestError(ctx, url, gctx, derr)
	}
	defer resp.Body.Close()
	data, rerr := io.ReadAll(resp.Body)
	if rerr != nil {
		return requestError(ctx, url, gctx, rerr)
	}
	m := value.NewMap()
	m.Set("status", value.Int(resp.StatusCode))
	m.Set("headers", headerMap(resp.Header))
	m.Set("body", value.Str(string(data)))
	return nat.Ok(m), nil
}

// requestError sorts a failed request: the task was cancelled, the
// deadline passed, or the transport failed.
func requestError(ctx value.Context, url string, gctx context.Context, err error) (value.Value, error) {
	select {
	case <-ctx.Done():
		return nil, diag.Errorf("E0701", diag.Pos{})
	default:
	}
	if errors.Is(gctx.Err(), context.DeadlineExceeded) || errors.Is(err, context.DeadlineExceeded) {
		return nat.ErrValue("timeout", "request to `"+url+"` timed out"), nil
	}
	return nat.ErrValue("network", "cannot request `"+url+"`: "+nat.Reason(err)), nil
}
```

```go
// Each request runs the handler on a fresh fung task via RunTask.
func httpServeFn(ctx value.Context, args []value.Value) (value.Value, error) {
	addr, err := nat.NeedStr("addr", args[0])
	if err != nil {
		return nil, err
	}
	if err := nat.NeedFn("handler", args[1]); err != nil {
		return nil, err
	}
	handler := args[1]
	if _, isNil := args[2].(value.Nil); !isNil {
		if err := nat.NeedFn("ready", args[2]); err != nil {
			return nil, err
		}
	}
	ln, lerr := net.Listen("tcp", addr)
	if lerr != nil {
		return nat.ErrValue("network", "cannot listen on `"+addr+"`: "+nat.Reason(lerr)), nil
	}
	if ready, ok := args[2].(value.Fn); ok {
		if _, cerr := ctx.Call(ready, []value.Value{value.Str(ln.Addr().String())}); cerr != nil {
			ln.Close()
			return nil, cerr
		}
	}
	srv := &http.Server{Handler: http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		serveOne(ctx, handler, w, r)
	})}
	go srv.Serve(ln)
	<-ctx.Done()
	shutdownCtx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
	defer cancel()
	srv.Shutdown(shutdownCtx)
	return nil, diag.Errorf("E0701", diag.Pos{})
}

// requestValue builds the handler's req map (06).
func requestValue(r *http.Request) (value.Value, error) {
	body, err := io.ReadAll(r.Body)
	if err != nil {
		return nil, err
	}
	query := value.NewMap()
	keys := make([]string, 0, len(r.URL.Query()))
	for k := range r.URL.Query() {
		keys = append(keys, k)
	}
	sort.Strings(keys)
	for _, k := range keys {
		query.Set(k, value.Str(r.URL.Query().Get(k)))
	}
	m := value.NewMap()
	m.Set("method", value.Str(r.Method))
	m.Set("path", value.Str(r.URL.Path))
	m.Set("query", query)
	m.Set("headers", headerMap(r.Header))
	m.Set("body", value.Str(string(body)))
	return m, nil
}

// serveOne runs the handler for one request and writes its answer. A
// handler error or an unusable result is a 500 reported on stderr; the
// server goes on (06).
func serveOne(ctx value.Context, handler value.Value, w http.ResponseWriter, r *http.Request) {
	req, err := requestValue(r)
	if err != nil {
		http.Error(w, "cannot read the request body", http.StatusBadRequest)
		return
	}
	res, cerr := ctx.RunTask(handler, []value.Value{req})
	if cerr != nil {
		if _, isCancel := cerr.(*diag.Error); isCancel && cerr.(*diag.Error).Code == "E0701" {
			http.Error(w, "server shutting down", http.StatusServiceUnavailable)
			return
		}
		io.WriteString(ctx.Stderr(), "http: handler failed: "+cerr.Error()+"\n")
		http.Error(w, "handler failed", http.StatusInternalServerError)
		return
	}
	status, headers, body, ok := responseParts(res)
	if !ok {
		io.WriteString(ctx.Stderr(), "http: handler returned "+res.TypeName()+", expected a str or a map\n")
		http.Error(w, "handler failed", http.StatusInternalServerError)
		return
	}
	for k, v := range headers {
		w.Header().Set(k, v)
	}
	if w.Header().Get("Content-Type") == "" && body != "" {
		w.Header().Set("Content-Type", "text/plain; charset=utf-8")
	}
	w.WriteHeader(status)
	io.WriteString(w, body)
}

// responseParts reads a handler result: a str is a 200 text answer, a
// map carries {status = 200, headers = {}, body = ""}.
func responseParts(res value.Value) (int, map[string]string, string, bool) {
	switch x := res.(type) {
	case value.Str:
		return 200, nil, string(x), true
	case *value.Map:
		status := 200
		if v, ok := x.Get("status"); ok {
			n, isInt := v.(value.Int)
			if !isInt || n < 100 || n > 599 {
				return 0, nil, "", false
			}
			status = int(n)
		}
		body := ""
		if v, ok := x.Get("body"); ok {
			s, isStr := v.(value.Str)
			if !isStr {
				return 0, nil, "", false
			}
			body = string(s)
		}
		headers := map[string]string{}
		if v, ok := x.Get("headers"); ok {
			hm, isMap := v.(*value.Map)
			if !isMap {
				return 0, nil, "", false
			}
			for _, k := range hm.Keys() {
				hv, _ := hm.Get(k)
				s, isStr := hv.(value.Str)
				if !isStr {
					return 0, nil, "", false
				}
				headers[k] = string(s)
			}
		}
		return status, headers, body, true
	}
	return 0, nil, "", false
}
```

```
# api: the handlers behind serve.fung — GET /health, POST /echo (echoes
# the JSON body back with a timestamp), anything else 404.
import http
import json
import time

fn json_response(status, data) =
  {status, headers: {"content-type": "application/json"}, body: json.stringify(data)}

fn health(started, requests) =
  json_response(200, {ok: true, uptime: str(time.now() - started), requests: deref(requests)})

fn echo(req):
  match json.parse(req.body):
    Ok(data) => json_response(200, {received: data, at: time.format(time.now())})
    Err(e) => {status: 400, body: "bad json: ${e.message}"}
  ;;
;;

# Builds the handler: the server's state lives in the closure (05: 공유
# 상태는 atom).
fn handler(started, requests) = fn(req):
  swap(requests, n => n + 1)
  match (req.method, req.path):
    ("GET", "/health") => health(started, requests)
    ("POST", "/echo") => echo(req)
    _ => {status: 404, body: "not found: ${req.path}"}
  ;;
;;

test "health, echo and 404":
  port = channel(1)
  h = handler(time.now(), atom(0))
  s = spawn(() => http.serve("127.0.0.1:0", h, ready: a => send(port, a)))
  base = "http://" + recv(port)!
  r = http.get("${base}/health")!
  assert r.status == 200
  assert json.parse(r.body)!.ok == true
  e = http.post("${base}/echo", json.stringify({x: 1}))!
  assert json.parse(e.body)!.received.x == 1
  assert http.post("${base}/echo", "{")!.status == 400
  assert http.get("${base}/nope")!.status == 404
  assert json.parse(http.get("${base}/health")!.body)!.requests == 4
  cancel(s)
;;
```

```
# urlcheck: fetches one URL and summarises the answer — used by
# checkurls.fung in parallel.
import http
import json
import regex
import time

# The top-level keys of a JSON body, [] for anything else.
fn json_keys(resp):
  ct = get(resp.headers, "content-type", "")
  if not regex.is_match(ct, "^application/json"): return [] ;;
  match json.parse(resp.body):
    Ok(m) when type_of(m) == 'map => keys(m)
    _ => []
  ;;
;;

fn check(url, timeout = 5s):
  t0 = time.now()
  r = http.get(url, timeout: timeout)
  took = time.now() - t0
  match r:
    Ok(resp) => {url, ok: resp.status < 400, status: resp.status, took, keys: json_keys(resp)}
    Err(e) => {url, ok: false, status: e.kind, took, keys: []}
  ;;
;;

# Prints one line per result and reports whether all were ok.
fn report(results):
  each(results, r => print(r.ok then "ok  " else "FAIL", r.status, r.took, r.url, join(r.keys, ",")))
  all(results, r => r.ok)
;;

test "check reports status and json keys":
  port = channel(1)
  s = spawn(() => http.serve("127.0.0.1:0", fn(req):
    req.path == "/j" then {headers: {"content-type": "application/json"}, body: `{"a":1,"b":2}`}
    else {status: 500, body: "x"}
  ;;, ready: a => send(port, a)))
  base = "http://" + recv(port)!
  good = check("${base}/j", 2s)
  assert good.ok
  assert good.keys == ["a", "b"]
  bad = check("${base}/x", 2s)
  assert not bad.ok
  assert bad.status == 500
  down = check("http://127.0.0.1:1/", 2s)
  assert down.status == 'network
  cancel(s)
;;
```
