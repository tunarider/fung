# Rust 이식 1단계: 앞단 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** fung의 앞단을 Rust로 옮겨, `fung dump ast|core|bytecode`의 출력·컴파일 에러·종료 코드를 Go 구현과 글자 단위로 같게 만든다.

**Architecture:** 저장소 루트를 Cargo 워크스페이스로 하고 `crates/fung-front`(앞단 라이브러리)와 `crates/fung-cli`(`dump`만 하는 `fung`)를 둔다. 단계 구성·균일 트리·명령 체계·에러 번호는 Go와 같고 각 모듈 안은 Rust 관용으로 쓴다. Go가 만든 `spec/builtins.txt`·`spec/errors.txt`를 Rust가 빌드할 때 읽고, Rust 테스트가 Go 바이너리를 빌드해 출력을 직접 비교한다.

**Tech Stack:** Rust 1.94 (edition 2024, 크레이트 0개), Go 1.27 (정답지·명세 생성).

**Spec:** `docs/superpowers/specs/2026-10-14-rust-front-end-design.md`

## 이식 규칙 (모든 Rust 작업에 적용)

이 계획의 Rust 코드는 **Go 소스가 참조 구현**이다. 각 작업은 대응하는 Go 파일을 함수 단위로 읽고 같은 동작을 Rust로 쓴다. 줄 단위 번역은 하지 않는다.
- 노드·토큰·명령 종류는 `enum`, 에러는 `Result<_, Diag>`/`Result<_, Vec<Diag>>`, 문자열은 입력은 `&str`·저장은 `String`, 반복은 반복자.
- 출력 순서가 드러나는 곳에서 `HashMap` 순회 금지(`BTreeMap`이나 `Vec` 사용).
- Go 단위 테스트는 각 작업에 이름으로 나열된 것을 **모두** Rust 테스트로 옮긴다(케이스 표까지). 작업마다 보인 테스트 코드는 그 모양의 본보기이자 반드시 들어갈 케이스다.
- Go에 있으나 범위 밖인 것: 경고(`internal/resolve/warn.go`, W01xx), REPL 옵션(`Globals`, `Redefine`), 런타임 값. 해당 Go 테스트(`TestWarnings`, `TestReplOptions`, `TestLoaderCollectsWarningsAndTests`의 경고 부분)는 옮기지 않는다.

```go
// tests/spec_test.go
package tests

import (
	"flag"
	"fmt"
	"os"
	"sort"
	"strings"
	"testing"

	"github.com/tunarider/fung/extras"
	"github.com/tunarider/fung/internal/diag"
	"github.com/tunarider/fung/internal/frontend"
	"github.com/tunarider/fung/internal/native"
	"github.com/tunarider/fung/internal/prelude"
	"github.com/tunarider/fung/internal/value"
	"github.com/tunarider/fung/internal/vm"
)

var update = flag.Bool("update", false, "rewrite spec/*.txt")

func params(ps []value.NativeParam) string {
	out := make([]string, len(ps))
	for i, p := range ps {
		switch {
		case p.Rest:
			out[i] = "..." + p.Name
		case p.Default != nil:
			out[i] = p.Name + "=?"
		default:
			out[i] = p.Name
		}
	}
	return strings.Join(out, ", ")
}

func builtinsSpec(t *testing.T) string {
	var lines []string
	natives := prelude.Core(nil)
	for k, v := range vm.Concurrency() {
		natives[k] = v
	}
	for name, v := range natives {
		n, ok := v.(*value.Native)
		if !ok {
			lines = append(lines, "prelude native "+name) // a non-function prelude value (args)
			continue
		}
		lines = append(lines, fmt.Sprintf("prelude native %s(%s)", name, params(n.Params)))
	}
	src, _ := extras.Source("prelude")
	b, errs := frontend.Compile("extras/prelude.fung", src, natives, frontend.StageAll)
	if len(errs) > 0 {
		t.Fatalf("compile prelude.fung: %v", errs[0])
	}
	for _, name := range b.Proto.Exports {
		lines = append(lines, "prelude fung "+name)
	}
	mods := map[string]*value.Module{}
	for k, m := range native.Bridges() {
		mods[k] = m
	}
	for k, m := range vm.NativeModules() {
		mods[k] = m
	}
	for key, m := range mods {
		for _, name := range m.Names() {
			v, _ := m.Get(name)
			if n, ok := v.(*value.Native); ok {
				lines = append(lines, fmt.Sprintf("module %s %s(%s)", key, name, params(n.Params)))
			} else {
				lines = append(lines, fmt.Sprintf("module %s %s", key, name))
			}
		}
	}
	sort.Strings(lines)
	return "# fung built-in names. Generated by `go test ./tests -run TestSpecFiles -update`; do not edit.\n" +
		"# prelude native <name>(<params>)   Go function in the prelude\n" +
		"# prelude fung <name>               public name of extras/prelude.fung\n" +
		"# module <_native.x> <name>(<params>)\n" + strings.Join(lines, "\n") + "\n"
}

func errorsSpec(t *testing.T) string {
	var b strings.Builder
	b.WriteString("# fung error and warning catalog. Generated by `go test ./tests -run TestSpecFiles -update`; do not edit.\n")
	b.WriteString("# code<TAB>message template<TAB>underline label<TAB>help\n")
	for _, code := range diag.Codes() {
		fields := []string{code, diag.Template(code), diag.Label(code), diag.Hint(code)}
		for _, f := range fields {
			if strings.ContainsAny(f, "\t\n") {
				t.Fatalf("%s: a catalog field contains a tab or newline: %q", code, f)
			}
		}
		b.WriteString(strings.Join(fields, "\t") + "\n")
	}
	return b.String()
}

// The spec files are the language-neutral record of built-in names and
// error messages; the Rust port reads them. They must match the Go code.
func TestSpecFiles(t *testing.T) {
	for path, want := range map[string]string{"../spec/builtins.txt": builtinsSpec(t), "../spec/errors.txt": errorsSpec(t)} {
		if *update {
			if err := os.WriteFile(path, []byte(want), 0o644); err != nil {
				t.Fatal(err)
			}
			continue
		}
		got, err := os.ReadFile(path)
		if err != nil || string(got) != want {
			t.Errorf("%s is out of date: run `go test ./tests -run TestSpecFiles -update`", path)
		}
	}
}
```

템플릿은 `include_str!("../../../../spec/errors.txt")`로 읽어 `OnceLock<BTreeMap<String, Entry>>`에 한 번 파싱한다(`Entry { message, label, help }`).

// crates/fung-front/src/diag.rs (아래쪽)
#[cfg(test)]
mod tests {
    use super::*;
    use crate::source::{Pos, SourceMap};

    fn at(sm: &mut SourceMap, name: &str, text: &str, line: u32, col: u32) -> Pos {
        let file = sm.add(name, text);
        Pos { file, line, col }
    }

    #[test]
    fn format_undefined_name_with_label() {
        let mut sm = SourceMap::new();
        let pos = at(&mut sm, "-e", "print(nope)", 1, 7);
        let d = error("E0204", pos, vec![Arg::Str("nope".into())]).span(4);
        assert_eq!(
            format(&d, &sm),
            "error[E0204]: undefined name `nope`\n  --> -e:1:7\n  |\n1 | print(nope)\n  |       ^^^^ not found in this scope\n  |\n"
        );
    }

    #[test]
    fn format_with_catalog_help() {
        let mut sm = SourceMap::new();
        let pos = at(&mut sm, "-e", "y = 1 < 2 < 3", 1, 11);
        let d = error("E0111", pos, vec![]);
        assert_eq!(
            format(&d, &sm),
            "error[E0111]: comparison operators cannot be chained\n  --> -e:1:11\n  |\n1 | y = 1 < 2 < 3\n  |           ^\n  |\n help: split it with `and`: a < b and b < c\n"
        );
    }

    #[test]
    fn excerpt_expands_tabs() {
        let mut sm = SourceMap::new();
        let pos = at(&mut sm, "-e", "\tx = \"\\q\"", 1, 7);
        let d = error("E0103", pos, vec![Arg::Str("q".into())]);
        assert_eq!(
            format(&d, &sm),
            "error[E0103]: unknown escape \\q in string\n  --> -e:1:7\n  |\n1 |     x = \"\\q\"\n  |          ^\n  |\n"
        );
    }

    #[test]
    fn columns_and_underline_count_chars() {
        let mut sm = SourceMap::new();
        let pos = at(&mut sm, "k.fung", "s = \"한글\"\nprint(없는)\n", 2, 7);
        let d = error("E0204", pos, vec![Arg::Str("없는".into())]).span(2);
        assert_eq!(
            format(&d, &sm),
            "error[E0204]: undefined name `없는`\n  --> k.fung:2:7\n  |\n2 | print(없는)\n  |       ^^ not found in this scope\n  |\n"
        );
    }

    #[test]
    fn crlf_and_bom_lines() {
        let mut sm = SourceMap::new();
        let pos = at(&mut sm, "crlf.fung", "x = 1\r\ny = )\r\n", 2, 5);
        let d = error("E0110", pos, vec![Arg::Str("`)`".into())]);
        assert_eq!(format(&d, &sm), "error[E0110]: unexpected `)`\n  --> crlf.fung:2:5\n  |\n2 | y = )\n  |     ^\n  |\n");
        let pos = at(&mut sm, "bom.fung", "\u{feff}x = )\n", 1, 5);
        let d = error("E0110", pos, vec![Arg::Str("`)`".into())]);
        assert_eq!(format(&d, &sm), "error[E0110]: unexpected `)`\n  --> bom.fung:1:5\n  |\n1 | \u{feff}x = )\n  |     ^\n  |\n");
    }

    #[test]
    fn quote_matches_go() {
        assert_eq!(quote("a\tb\"c "), "\"a\\tb\\\"c \"");
        assert_eq!(quote("\u{0}"), "\"\\x00\"");
        assert_eq!(quote("한"), "\"한\"");
    }

    #[test]
    fn verbs() {
        assert_eq!(message("E0101", &[Arg::Str("@".into())]), "unexpected character \"@\"");
        assert!(is_compile_error("E0110") && is_compile_error("E0204") && is_compile_error("E0301"));
        assert!(!is_compile_error("E0502"));
    }
}

```rust
// crates/fung-cli/tests/parity.rs
use std::{path::{Path, PathBuf}, process::Command};

fn repo() -> PathBuf {
    Path::new(env!("CARGO_MANIFEST_DIR")).join("../..").canonicalize().unwrap()
}

fn go_fung() -> Option<PathBuf> {
    let out = std::env::temp_dir().join(format!("fung-go-parity-{}", std::process::id()));
    let ok = Command::new("go").args(["build", "-o"]).arg(&out).arg("./cmd/fung")
        .current_dir(repo()).status().map(|s| s.success()).unwrap_or(false);
    ok.then_some(out)
}

fn corpus() -> Vec<String> {
    let mut files = Vec::new();
    for dir in ["tests/lang", "examples", "extras", "bench"] {
        let mut stack = vec![repo().join(dir)];
        while let Some(d) = stack.pop() {
            for e in std::fs::read_dir(&d).unwrap() {
                let p = e.unwrap().path();
                if p.is_dir() { stack.push(p); }
                else if p.extension().is_some_and(|x| x == "fung") {
                    files.push(p.strip_prefix(repo()).unwrap().to_string_lossy().into_owned());
                }
            }
        }
    }
    files.sort();
    files
}

fn run(bin: &Path, args: &[&str]) -> (String, String, i32) {
    let o = Command::new(bin).args(args).current_dir(repo()).output().unwrap();
    (String::from_utf8_lossy(&o.stdout).into(), String::from_utf8_lossy(&o.stderr).into(), o.status.code().unwrap_or(-1))
}

fn first_diff(a: &str, b: &str) -> String {
    for (i, (x, y)) in a.lines().zip(b.lines()).enumerate() {
        if x != y { return format!("line {}:\n  go:   {x}\n  rust: {y}", i + 1); }
    }
    format!("lengths differ: go {} lines, rust {} lines", a.lines().count(), b.lines().count())
}

#[test]
fn dumps_match_the_go_implementation() {
    let Some(go) = go_fung() else {
        eprintln!("skipping parity: `go` is not available");
        return;
    };
    let rust = PathBuf::from(env!("CARGO_BIN_EXE_fung"));
    let mut failures = Vec::new();
    let mut cases: Vec<Vec<String>> = Vec::new();
    for f in corpus() {
        for stage in ["ast", "core", "bytecode"] {
            cases.push(vec!["dump".into(), stage.into(), f.clone()]);
        }
    }
    for extra in [
        vec!["dump"], vec!["dump", "nope"], vec!["dump", "core"], vec!["dump", "core", "-e"],
        vec!["dump", "core", "no/such/file.fung"], vec!["dump", "bytecode", "-e", "x = )"],
    ] {
        cases.push(extra.into_iter().map(String::from).collect());
    }
    let deep = format!("x = {}1{}", "(".repeat(20_000), ")".repeat(20_000));
    cases.push(vec!["dump".into(), "ast".into(), "-e".into(), deep]);
    for args in &cases {
        let a: Vec<&str> = args.iter().map(String::as_str).collect();
        let (g, r) = (run(&go, &a), run(&rust, &a));
        if g != r {
            let what = if g.0 != r.0 { first_diff(&g.0, &r.0) } else if g.1 != r.1 { first_diff(&g.1, &r.1) } else { format!("exit {} vs {}", g.2, r.2) };
            failures.push(format!("{}: {what}", a.iter().take(3).cloned().collect::<Vec<_>>().join(" ")));
        }
    }
    let _ = std::fs::remove_file(&go);
    assert!(failures.is_empty(), "{} of {} cases differ:\n{}", failures.len(), cases.len(), failures.join("\n"));
}
```
