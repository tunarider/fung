| # dotenv: read KEY=VALUE files and check them against requirements.
|
| # Text → map of KEY → VALUE. Blank lines and # comments are skipped;
| # a `export ` prefix and surrounding quotes are removed.
| fn parse(text):
|   rows = lines(text) |> map(trim) |> filter(l => l != "" and not starts_with(l, "#"))
|   fold(rows, {}, fn(acc, line):
|     (k, v) = match split(line, "="):

|     ;;

==== extras/cli.fung
-- stdout
| # cli: command-line argument parsing (06). Small on purpose: flags and the
| # rest; checking and converting values is ordinary code.
| #
| #   cli.parse(["in", "--out", "x", "--", "-v"])
| #   → {flags: {out: "x"}, rest: ["in", "-v"]}
| #
| # `--name value` and `--name=value` set a flag; `--name` alone, or a name
| # listed in `bools`, is true. Flag names turn `-` into `_` (`dry_run`).
| # Arguments not starting with `--` go to rest in order; everything after
| # a bare `--` goes to rest. A lone `-` is an ordinary argument.
| fn parse(argv, bools = []) = _parse(argv, bools, {flags: {}, rest: []})
|
| fn _parse(argv, bools, acc) = match argv:
|   []                                     => acc
|   ["--", ...rest]                        => put(acc, "rest", acc.rest + rest)
|   [a, ...rest] when starts_with(a, "--") =>
|     (k, v, left) = _flag(a[2..], rest, bools)
|     _parse(left, bools, put(acc, "flags", put(acc.flags, k, v)))
|   [a, ...rest]                           => _parse(rest, bools, put(acc, "rest", acc.rest + [a]))
| ;;
|
| # One flag body (after `--`) with the remaining arguments → (key, value, rest).
| fn _flag(body, rest, bools) = match split(body, "="):
|   [name, ...vals] when not is_empty(vals) => (_key(name), join(vals, "="), rest)
|   _                                       => body in bools then (_key(body), true, rest)
|     else match rest:
|       [v, ...more] when not starts_with(v, "--") => (_key(body), v, more)
|       _                                          => (_key(body), true, rest)
|     ;;
| ;;
|
| fn _key(name) = replace(name, "-", "_")
|

| # Opens path, calls f with the handle, and always closes it (03: a
| # `with`). f's result — a value or the Err that `?` returned — is passed
| # through.
| #
| # The handle is opened inside the `with` (05), so nothing can come between
| # opening and the promise to close it. A close that fails (a write that
| # could not be flushed) is a runtime error; a handle f closed itself is
| # not a failure.
| fn with_open(path, f, mode = 'read) = with(() => _n.open(path, mode), r => _with_ok(r, f), r => _close_opened(r))
|
| # Creates a temporary directory, calls f with its path, and removes it,
| # whatever f does (a `with`, made inside it as `with_open` opens). A
| # removal that fails is a runtime error.
| fn with_temp_dir(f) = with(() => _n.temp_dir(), r => _with_ok(r, f), r => _remove_made(r))
|
| # f of what was acquired, or the Err acquiring gave.
| fn _with_ok(r, f) = match r:
|   Ok(x) => f(x)
|   e     => e

|   Ok(h) => match _n.close(h):

|   _     => nil
| ;;

|   Ok(d) => _n.remove_all(d)!
